```typescript
async compress(observations: any[]): Promise<HarmonicUnit> {
  if (observations.length === 0) {
    return this.makeUnit(observations, [], 'semantic', 'empty', 0);
  }

  const highEnergy = observations.filter(o => (o.energy ?? 0.5) > this.energyThreshold);
  const useHighEnergy = highEnergy.length > 0 ? highEnergy : observations;
  const memoryType = this.detectMemoryType(useHighEnergy);
  const energy = useHighEnergy.reduce((s, o) => s + (o.energy ?? 0.5), 0) / useHighEnergy.length;

  // Try LLM via Gateway
  try {
    const res = await fetch(`${this.gatewayUrl}/api/llm/compress`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ observations: useHighEnergy.map(o => o.content || '') }),
      signal: AbortSignal.timeout(10000)
    });
    if (res.ok) {
      const result = await res.json();
      const unit = this.makeUnit(useHighEnergy, result.concepts || [], memoryType,
        result.narrative?.slice(0, 50) || 'compressed', result.energy ?? energy);
      unit.memory_value = result.narrative || useHighEnergy.map(o => o.content).join('\n');
      unit.cue_anchors = result.concepts || [];
      await this.persistUnit(unit, memoryType);
      return unit;
    }
  } catch {}

  // Fallback: rule-based
  const zeroResult = this.zeroToken.compress(useHighEnergy);
  const allFacts = [];
  const allConcepts = [];
  const conceptPatterns = [
    { regex: /(jwt|oauth|bcrypt|auth)/i, concept: 'authentication' },
    { regex: /(error|fail|exception)/i, concept: 'error' },
    { regex: /(coverage|test|jest)/i, concept: 'testing' },
    { regex: /(api|endpoint|route)/i, concept: 'api' },
    { regex: /(config|setting|env)/i, concept: 'configuration' }
  ];
  for (const obs of useHighEnergy) {
    const content = obs.content || '';
    for (const { regex, concept } of conceptPatterns) {
      if (regex.test(content)) allConcepts.push(concept);
    }
  }
  const unit = this.makeUnit(useHighEnergy, [...new Set(allConcepts)], memoryType,
    useHighEnergy[0]?.content?.slice(0, 50) || 'rule-compressed', energy);
  unit.memory_value = useHighEnergy.map(o => o.content).join('\n').slice(0, 1000);
  await this.persistUnit(unit, memoryType);
  return unit;
}

private detectMemoryType(observations: any[]): 'episodic' | 'semantic' | 'procedural' {
  const allContent = observations.map(o => o.content || '').join(' ');
  const isFileEdit = observations.every(o => o.type === 'file_edit' || (o.content && o.content.includes('File modified')));
  if (isFileEdit) return 'procedural';
  if (/plan|wave|task|goal/i.test(allContent)) return 'episodic';
  return 'semantic';
}

private makeUnit(
  observations: any[], concepts: string[], memoryType: string,
  abstraction: string, energy: number
): HarmonicUnit {
  return {
    id: generateHarmonicId(),
    goal_id: this.extractGoalId(observations),
    memory_type: memoryType as any,
    primary_abstraction: abstraction.slice(0, 50),
    cue_anchors: concepts.slice(0, 8),
    memory_value: '',
    energy: Math.max(0, Math.min(1, energy)),
    created_at: new Date().toISOString(),
    updated_at: new Date().toISOString()
  };
}

private async persistUnit(unit: HarmonicUnit, memoryType: string): Promise<void> {
  if (!this.baseDir || !this.harmonicIndex) return;
  const tier = memoryType === 'procedural' ? 'tier4' : memoryType === 'episodic' ? 'tier2' : 'tier3';
  const goalId = unit.goal_id || '__global__';
  const filePath = path.join(this.baseDir, 'memory', tier, `${goalId}.json`);
  const dir = path.dirname(filePath);
  if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
  const existing = fs.existsSync(filePath) ? JSON.parse(fs.readFileSync(filePath, 'utf-8')) : [];
  existing.push(unit);
  fs.writeFileSync(filePath, JSON.stringify(existing, null, 2), 'utf-8');
  this.harmonicIndex.addEntry(unit, tier);
}

private extractGoalId(observations: any[]): string | null {
  for (const o of observations) {
    if (o.goalId) return o.goalId;
    if (o.metadata?.goalId) return o.metadata.goalId;
  }
  return null;
}
```

- [ ] **Step 2: Write `src/memory/minhash-merger.ts`**

```typescript
import * as fs from 'fs';
import * as path from 'path';
import { HarmonicUnit } from './harmonic-types';
import { HarmonicIndexManager } from './harmonic-index';

export class MinHashMerger {
  private signatureSize: number = 4;
  private threshold: number = 0.6;

  generateSignature(text: string): number[] {
    const shingles = this.extractShingles(text.toLowerCase());
    const sig = new Array(this.signatureSize).fill(Infinity);
    for (const shingle of shingles) {
      for (let i = 0; i < this.signatureSize; i++) {
        const hash = this.hashFunction(shingle, i);
        if (hash < sig[i]) sig[i] = hash;
      }
    }
    return sig;
  }

  similarity(sigA: number[], sigB: number[]): number {
    if (sigA.length !== sigB.length) return 0;
    let matches = 0;
    for (let i = 0; i < sigA.length; i++) {
      if (sigA[i] === sigB[i]) matches++;
    }
    return matches / sigA.length;
  }

  async merge(
    unit: HarmonicUnit,
    tier: string,
    indexManager: HarmonicIndexManager,
    baseDir: string
  ): Promise<HarmonicUnit> {
    const allEntries = indexManager.getIndex().entries;
    const newSig = this.generateSignature(unit.primary_abstraction);

    for (const entry of allEntries) {
      if (entry.id === unit.id) continue;
      const existingSig = this.generateSignature(entry.primary_abstraction);
      if (this.similarity(newSig, existingSig) > this.threshold) {
        const existingUnit = this.loadUnit(baseDir, entry.tier, entry.id);
        if (!existingUnit) continue;

        unit.primary_abstraction = (unit.primary_abstraction + ' + ' + existingUnit.primary_abstraction).slice(0, 80);
        unit.cue_anchors = [...new Set([...unit.cue_anchors, ...existingUnit.cue_anchors])].slice(0, 8);
        unit.memory_value = unit.memory_value + '\n---\n' + existingUnit.memory_value;
        unit.energy = Math.min(1, unit.energy + 0.15);
        unit.merged_from = [...(unit.merged_from || []), existingUnit.id];
        unit.updated_at = new Date().toISOString();

        indexManager.removeEntry(entry.id);
      }
    }
    return unit;
  }

  private loadUnit(baseDir: string, tier: string, id: string): HarmonicUnit | null {
    const dir = path.join(baseDir, 'memory', tier);
    if (!fs.existsSync(dir)) return null;
    const files = fs.readdirSync(dir).filter(f => f.endsWith('.json'));
    for (const file of files) {
      try {
        const units: HarmonicUnit[] = JSON.parse(fs.readFileSync(path.join(dir, file), 'utf-8'));
        const found = units.find(u => u.id === id);
        if (found) return found;
      } catch {}
    }
    return null;
  }

  private extractShingles(text: string): string[] {
    const shingles: string[] = [];
    for (let i = 0; i < text.length - 2; i++) {
      shingles.push(text.slice(i, i + 3));
    }
    return shingles;
  }

  private hashFunction(shingle: string, seed: number): number {
    let hash = seed;
    for (let i = 0; i < shingle.length; i++) {
      hash = ((hash << 5) - hash) + shingle.charCodeAt(i);
      hash |= 0;
    }
    return Math.abs(hash);
  }
}
```

import * as fs from 'fs';
import * as path from 'path';
import os from 'os';
import { HarmonicIndexManager } from '../../src/memory/harmonic-index';
import { HarmonicUnit } from '../../src/memory/harmonic-types';

describe('Harmonic Memory E2E', () => {
  let tmpDir: string;
  let indexManager: HarmonicIndexManager;
  let merger: MinHashMerger;

  beforeEach(() => {
    tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), 'hm-e2e-'));

    indexManager = new HarmonicIndexManager(tmpDir);
    merger = new MinHashMerger();
  });

  afterEach(() => {
    fs.rmSync(tmpDir, { recursive: true, force: true });
  });

  it('full flow: write → index → search', () => {
    const unit: HarmonicUnit = {
      id: 'e2e_1', goal_id: 'g1', memory_type: 'semantic',
      primary_abstraction: 'JWT token expiry config',
      cue_anchors: ['jwt', 'auth'],
      memory_value: 'JWT should expire in 30 minutes',
      energy: 0.8, created_at: '', updated_at: ''
    };
    // Write tier file
    const tierDir = path.join(tmpDir, 'memory', 'tier3');
    fs.mkdirSync(tierDir, { recursive: true });
    fs.writeFileSync(path.join(tierDir, 'g1.json'), JSON.stringify([unit]), 'utf-8');
    // Add to index
    indexManager.addEntry(unit, 'tier3');

    // Search
    const results = indexManager.search('JWT');
    expect(results.length).toBeGreaterThan(0);
    expect(results[0].id).toBe('e2e_1');
  });

  it('search is tier-agnostic', () => {
    const u1: HarmonicUnit = { id: 'e2e_2', goal_id: 'g1', memory_type: 'semantic', primary_abstraction: 'API rate limit', cue_anchors: ['api'], memory_value: 'Rate limit 100 req/min', energy: 0.7, created_at: '', updated_at: '' };

    fs.writeFileSync(path.join(tmpDir, 'memory', 'tier2', 'g1.json'), JSON.stringify([u2]), 'utf-8');
    indexManager.addEntry(u1, 'tier3');
    indexManager.addEntry(u2, 'tier2');
