export function calculateSalience(text: string): number {
  if (!text) return 1.0;
  const HIGH = [/error|crash|fail|丢失|数据损坏|宕机|critical|emergency|严重/i];
  const LOW = [/info|success|完成|正常|debug|trace|verbose/i];
  if (HIGH.some(p => p.test(text))) return 1.5;
  if (LOW.some(p => p.test(text))) return 0.5;
  return 1.0;
}

```typescript
import * as fs from 'fs';
import * as path from 'path';

export interface CognitiveEdge {
  source: string;
  target: string;
  weight: number;
}

export interface CognitiveGraph {
  version: number;
  updated_at: string;
  edges: CognitiveEdge[];
}

export class CognitiveGraphManager {
  private path: string;
  private graph: CognitiveGraph;

  constructor(baseDir: string) {
    this.path = path.join(baseDir, 'memory', '.cognitive_graph.json');
    this.graph = this.load();
  }

  private load(): CognitiveGraph {
    try {
      if (fs.existsSync(this.path)) {
        return JSON.parse(fs.readFileSync(this.path, 'utf-8'));
      }
    } catch {}
    return { version: 1, updated_at: new Date().toISOString(), edges: [] };
  }

  private save(): void {
    this.graph.updated_at = new Date().toISOString();
    const tmpPath = this.path + '.tmp';
    fs.writeFileSync(tmpPath, JSON.stringify(this.graph, null, 2), 'utf-8');
    fs.renameSync(tmpPath, this.path);
  }

  addConnection(idA: string, idB: string): void {
    if (idA === idB) return;
    const [src, tgt] = [idA, idB].sort();
    const existing = this.graph.edges.find(e => e.source === src && e.target === tgt);
    if (existing) {
      existing.weight++;
    } else {
      this.graph.edges.push({ source: src, target: tgt, weight: 1 });
    }
    this.save();
  }

  getTopAssociations(id: string, topK: number = 2): string[] {
    return this.graph.edges
      .filter(e => e.source === id || e.target === id)
      .sort((a, b) => b.weight - a.weight)
      .slice(0, topK)
      .map(e => e.source === id ? e.target : e.source);
  }

  prune(threshold: number = 2): void {
    this.graph.edges = this.graph.edges.filter(e => e.weight >= threshold);
    this.save();
  }

  getGraph(): CognitiveGraph {
    return { ...this.graph, edges: [...this.graph.edges] };
  }
}
```

import * as fs from 'fs';
import * as path from 'path';
import { HarmonicIndexManager } from './harmonic-index';

export function getNextInterval(reviewCount: number): number {
  return 1 * Math.pow(2, reviewCount);
}

export interface ReviewTask {
  unitId: string;
  primaryAbstraction: string;
  overdue: number;
}

export class ReviewScheduler {
  private indexManager: HarmonicIndexManager;
  private baseDir: string;
  private interval: NodeJS.Timeout | null = null;

  constructor(indexManager: HarmonicIndexManager, baseDir: string) {
    this.indexManager = indexManager;
    this.baseDir = baseDir;
  }

  start(periodMs: number = 3600000): void {
    this.stop();
    this.interval = setInterval(() => this.tick(), periodMs);
  }

  stop(): void {
    if (this.interval) { clearInterval(this.interval); this.interval = null; }
  }

  getReviewQueue(): ReviewTask[] {
    const qPath = path.join(this.baseDir, 'memory', '.review_queue.json');
    try { return JSON.parse(fs.readFileSync(qPath, 'utf-8')).tasks || []; } catch { return []; }
  }

  private tick(): void {
    try {
      const now = Date.now();
      const entries = this.indexManager.getIndex().entries;
      const tasks: ReviewTask[] = [];

      for (const e of entries) {
        if (e.energy < 0.5) continue;
        const lastReview = e.last_reviewed ? new Date(e.last_reviewed).getTime() : now;
        const daysSince = (now - lastReview) / 86400000;
        const needed = getNextInterval(e.review_count || 0);
        if (daysSince >= needed) {
          tasks.push({ unitId: e.id, primaryAbstraction: e.primary_abstraction, overdue: Math.round(daysSince - needed) });
        }
      }

      if (tasks.length > 0) {
        tasks.sort((a, b) => b.overdue - a.overdue);
        const top = tasks.slice(0, 3);
        const qPath = path.join(this.baseDir, 'memory', '.review_queue.json');
        fs.writeFileSync(qPath, JSON.stringify({ generated_at: new Date().toISOString(), tasks: top }, null, 2), 'utf-8');
      }
    } catch {}
  }
}
```

import * as fs from 'fs';
import * as path from 'path';
import { HarmonicUnit, generateHarmonicId } from './harmonic-types';
import { HarmonicIndexManager } from './harmonic-index';

export interface DistillationResult {
  created: number;
  locked: number;
  errors: string[];
}

export async function runDistillation(
  indexManager: HarmonicIndexManager,
  baseDir: string
): Promise<DistillationResult> {
  const result: DistillationResult = { created: 0, locked: 0, errors: [] };
  const allEntries = indexManager.getIndex().entries;

  // Rule 1: T2 → T3 (≥3 similar episodic units)
  const t2Entries = allEntries.filter(e => e.tier === 'tier2' && e.memory_type === 'episodic');
  const groups = new Map<string, typeof t2Entries>();
  for (const e of t2Entries) {
    const key = e.primary_abstraction.toLowerCase().split(/\s+/).sort().slice(0, 3).join(' ');
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key)!.push(e);
  }

  for (const [, group] of groups) {
    if (group.length < 3) continue;
    try {
      const distilled: HarmonicUnit = {
        id: generateHarmonicId(),
        goal_id: group[0].goal_id || null,
        memory_type: 'semantic',
        abstraction_level: 2,
        primary_abstraction: [...new Set(group.map(g => g.primary_abstraction))].join(' | ').slice(0, 80),
        cue_anchors: [...new Set(group.flatMap(g => g.cue_anchors))].slice(0, 8),
        memory_value: group.map(g => g.memory_value).join('\n---\n'),
        energy: Math.min(1, group.reduce((s: number, g: any) => s + (g.energy || 0.5), 0) / group.length + 0.1),
        created_at: new Date().toISOString(),
        updated_at: new Date().toISOString(),
        merged_from: group.map(g => g.id)
      };

      const memoryDir = path.join(baseDir, 'memory', 'tier3');
      if (!fs.existsSync(memoryDir)) fs.mkdirSync(memoryDir, { recursive: true });
      const tier3File = path.join(memoryDir, `${distilled.goal_id || '__global__'}.json`);
      const existing = fs.existsSync(tier3File) ? JSON.parse(fs.readFileSync(tier3File, 'utf-8')) : [];
      existing.push(distilled);
      fs.writeFileSync(tier3File, JSON.stringify(existing, null, 2), 'utf-8');
      indexManager.addEntry(distilled, 'tier3');

      // Lock old T2s
      for (const old of group) {
        indexManager.updateEnergy(old.id, -0.5);
      }

      result.created++;
      result.locked += group.length;
    } catch (err: any) {
      result.errors.push(err.message);
    }
  }

  return result;
}
```
