### 1. Real LLM Compression via Gateway Proxy

#### Architecture

```
Plugin (LLMCompressor)
    │  POST /api/llm/compress { observations, model? }
    ▼
Gateway (llmCompress handler)
    │  Call Anthropic/OpenAI API with compression prompt
    ▼
LLM Provider → { narrative, facts, concepts, energy }
    │
    ▼
Plugin receives → CompressedMemory
    │  (Fallback to rule-based pattern matching if Gateway unreachable)
```

#### LLM Compression Prompt

```typescript
private async llmCompress(observations: string[], model?: string): Promise<any> {
  const config = this.loadLLMConfig();
  const provider = config.compression.provider;
  const apiKey = process.env[config.compression.apiKeyEnv];
  const modelName = model || config.compression.model;

  const prompt = `Analyze the following agent observations and extract structured memories.
Return JSON only:
{
  "narrative": "summary of what happened",
  "facts": ["specific fact 1", "specific fact 2"],
  "concepts": ["keyword1", "keyword2"],
  "energy": 0.5
}

Observations:
${observations.map((o, i) => `[${i + 1}] ${o}`).join('\n')}`;

  if (provider === 'anthropic') {
    // Call Anthropic Messages API
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': apiKey,
        'anthropic-version': '2023-06-01'
      },
      body: JSON.stringify({
        model: modelName,
        max_tokens: 500,
        messages: [{ role: 'user', content: prompt }]
      })
    });
    const data = await res.json();
    return this.parseLLMResponse(data.content?.[0]?.text || '');
  }

  if (provider === 'openai') {
    const res = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${apiKey}`
      },
      body: JSON.stringify({
        model: modelName,
        max_tokens: 500,
        messages: [{ role: 'user', content: prompt }]
      })
    });
    const data = await res.json();
    return this.parseLLMResponse(data.choices?.[0]?.message?.content || '');
  }

  throw new Error(`Unknown LLM provider: ${provider}`);
}
```

#### Plugin Changes

**File**: `src/compression/rule-based-compressor.ts` (modify)

Rename class to `HybridCompressor`. Add HTTP call to Gateway before falling back to rules:

```typescript
export class HybridCompressor {
  private gatewayUrl: string;

  constructor(gatewayUrl?: string) {
    this.gatewayUrl = gatewayUrl || 'http://127.0.0.1:3001';
  }

  async compress(observations: any[]): Promise<CompressedMemory> {
    const highEnergy = observations.filter(o => (o.energy ?? 0.5) > 0.6);
    if (highEnergy.length === 0) return this.ruleFallback(observations);

    // Try real LLM via Gateway
    try {
      const res = await fetch(`${this.gatewayUrl}/api/llm/compress`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          observations: highEnergy.map(o => o.content || '')
        }),
        signal: AbortSignal.timeout(10000)
      });
      if (res.ok) return res.json();
    } catch {}

    // Fallback: rule-based extraction
    return this.ruleFallback(highEnergy);
  }
}
```

window.updateWeight = function(key, val) {
  const data = app.state.alignment || { weights: { speed: 0.3, quality: 0.5, cost: 0.2 } };
  data.weights[key] = parseFloat(val);
  // Normalize to sum=1
  const total = Object.values(data.weights).reduce((a, b) => a + b, 0);

### 2. SCORE-driven State Machine

**File**: `src/engine/loop-state-machine.ts` (modify)

In `parseVerdict()`, if review content has a `score` field, use it:

```typescript
private async parseVerdict(): Promise<void> {
  if (this.pendingData && this.pendingData.verdict) {
    this.verdict = this.pendingData.verdict;
  } else {
    try {
      const reviewContent = await loadReview(this.goalId, this.loopNum, this.configDir);
      if (!reviewContent) { this.verdict = 'FAIL'; return; }

      // Try extracting score first
      const scoreMatch = reviewContent.match(/score[:\s]+(\d+)/i);
      if (scoreMatch) {
        const score = parseInt(scoreMatch[1], 10);
        if (score >= 85) this.verdict = 'PASS';
        else if (score >= 60) this.verdict = 'PARTIAL';
        else this.verdict = 'FAIL';
        return;
      }

      // Fallback to string matching
      const verdictMatch = reviewContent.match(/Verdict:\s*(PASS|FAIL|PARTIAL)/i);
      if (verdictMatch) { this.verdict = verdictMatch[1].toUpperCase(); }
      else if (reviewContent.includes('PASS')) { this.verdict = 'PASS'; }
      else { this.verdict = 'FAIL'; }
    } catch { this.verdict = 'FAIL'; }
  }
  await this.handleEvent('auto');
}
```

    if (prevReview && (prevReview.includes('FAIL') || prevReview.includes('score'))) {

### 1. Checkpoint Filename Sort

**File**: `gateway/src/recovery.ts` (modify)

```typescript
findLastCheckpoint(goalId: string): string | null {
  const checkpointsDir = path.join(this.projectDir, '.opencode/mafw/checkpoints', goalId);
  if (!fs.existsSync(checkpointsDir)) return null;

  const files = fs.readdirSync(checkpointsDir)
    .filter(f => f.endsWith('.json'))
    .sort((a, b) => {
      const numA = parseInt(a.match(/(\d+)\.json$/)![1], 10);
      const numB = parseInt(b.match(/(\d+)\.json$/)![1], 10);
      return numB - numA;
    });

  return files.length > 0 ? path.join(checkpointsDir, files[0]) : null;
}
```

### 4. SQLite Cost Storage

**File**: `src/plugin.ts` (modify `persistCosts`)

Write to both SQLite (via StorageBackend) and JSON (for Dashboard FS fallback):

```typescript
function persistCosts(): void {
  const all = costEstimator.getAllRecords();
  if (all.length === 0) return;

  // Write to SQLite via batch
  const storage = new (require('./storage/sqlite-storage').SQLiteStorage)(
    path.join(mafwDir, 'data', 'state.db')
  );
  storage.batch(all.map(r => ({
    type: 'set' as const,
    scope: 'cost_logs',
    key: r.id,
    value: r
  })));

  // Write JSON for Dashboard FS fallback
  const costDir = path.join(mafwDir, 'cost');
  if (!fs.existsSync(costDir)) fs.mkdirSync(costDir, { recursive: true });
  const byGoal = new Map<string, any[]>();
  for (const r of all) {
    const list = byGoal.get(r.goalId) || [];
    list.push(r);
    byGoal.set(r.goalId, list);
  }
  for (const [goalId, records] of byGoal) {
    fs.writeFileSync(path.join(costDir, `${goalId}.json`), JSON.stringify(records, null, 2), 'utf-8');
  }
}
```

|---|---|---|
| 1 | LLM via Gateway proxy, not direct Plugin call | API key isolation, single config point |

| 4 | SCORE drives state machine at ≥85/≥60/<60 | Reuses existing PARTIAL path for 60-84 |
| 5 | Plan reflection loads previous review | No new Agent, minimal code |
