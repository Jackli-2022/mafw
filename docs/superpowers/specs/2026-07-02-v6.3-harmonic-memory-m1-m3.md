# MAFW v6.3 Harmonic Memory M1-M3 Design

> **Goal**: Implement the first 3 phases of the Harmonic Memory system — unified data model (M1), migration (M2), and search (M3).

## Architecture Overview

```
v6.1 (current)
  │  T2/T3/T4 each have different JSON schemas
  │  Search requires knowing which tier to query
  │  BM25 + vector dual index
  ▼
v6.3 (M1-M3)
  │  All tiers share ONE data model: HarmonicUnit
  │  .harmonic_index.json as unified BM25 index
  │  Search is tier-agnostic
  │  Vector dependency removed
```

### Key Principles

- **YAGNI**: Don't implement MinHash merging (M5) or cognitive graph (v6.4) yet
- **Safe migration**: Old files backed up, not deleted
- **Idempotent**: Migration can run multiple times safely
- **No new dependencies**: BM25 implementation stays local (already in codebase)

### Index Manager (`src/memory/harmonic-index.ts`)

```typescript
export class HarmonicIndexManager {
  private indexPath: string;
  private index: HarmonicIndex;

  constructor(baseDir: string) {
    this.indexPath = path.join(baseDir, 'memory', '.harmonic_index.json');
    this.index = this.load();
  }

  private load(): HarmonicIndex {
    try {
      const raw = fs.readFileSync(this.indexPath, 'utf-8');
      return JSON.parse(raw);
    } catch {
      return { version: 1, updated_at: new Date().toISOString(), entries: [] };
    }
  }

  private save(): void {
    this.index.updated_at = new Date().toISOString();
    const tmpPath = this.indexPath + '.tmp';
    fs.writeFileSync(tmpPath, JSON.stringify(this.index, null, 2), 'utf-8');
    fs.renameSync(tmpPath, this.indexPath);
  }

  addEntry(unit: HarmonicUnit, tier: string): void {
    this.index.entries.push({
      id: unit.id,
      primary_abstraction: unit.primary_abstraction,
      cue_anchors: unit.cue_anchors,
      memory_type: unit.memory_type,
      goal_id: unit.goal_id,
      tier,
      energy: unit.energy
    });
    this.save();
  }

  removeEntry(id: string): void {
    this.index.entries = this.index.entries.filter(e => e.id !== id);
    this.save();
  }

  updateEnergy(id: string, delta: number): void {
    const entry = this.index.entries.find(e => e.id === id);
    if (entry) {
      entry.energy = Math.max(0, Math.min(1, entry.energy + delta));
      this.save();
    }
  }

  search(query: string, topK: number = 20): HarmonicIndexEntry[] {
    const tokens = query.toLowerCase().split(/\s+/).filter(t => t.length > 1);
    if (tokens.length === 0) return [];

    const scored = this.index.entries.map(entry => {
      const text = (entry.primary_abstraction + ' ' + entry.cue_anchors.join(' ')).toLowerCase();
      let score = 0;
      for (const token of tokens) {
        const regex = new RegExp(token.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi');
        const matches = text.match(regex);
        if (matches) score += matches.length;
      }
      return { entry, score: score * entry.energy };
    });

    return scored
      .filter(s => s.score > 0)
      .sort((a, b) => b.score - a.score)
      .slice(0, topK)
      .map(s => s.entry);
  }

  getIndex(): HarmonicIndex {
    return { ...this.index, entries: [...this.index.entries] };
  }
}
```

### Migration Logic (`src/memory/migrate-v6.1.ts`)

```typescript
export interface MigrationResult {
  migrated: number;
  skipped: number;
  errors: string[];
}

export async function migrateV61(
  baseDir: string,
  indexManager: HarmonicIndexManager
): Promise<MigrationResult> {
  const result: MigrationResult = { migrated: 0, skipped: 0, errors: [] };
  const memoryDir = path.join(baseDir, 'memory');

  // Ensure new directory structure
  for (const dir of ['tier2', 'tier3', 'tier4', 'l5']) {
    const d = path.join(memoryDir, dir);
    if (!fs.existsSync(d)) fs.mkdirSync(d, { recursive: true });
  }

  // 1. Migrate lessons/ → tier2/ (episodic)
  const lessonsDir = path.join(baseDir, 'lessons');
  if (fs.existsSync(lessonsDir)) {
    for (const file of fs.readdirSync(lessonsDir).filter(f => f.endsWith('.yml') || f.endsWith('.yaml') || f.endsWith('.json'))) {
      try {
        const content = fs.readFileSync(path.join(lessonsDir, file), 'utf-8');
        const unit: HarmonicUnit = {
          id: generateHarmonicId(),
          goal_id: file.replace(/\.(yml|yaml|json)$/, '').split('-')[0] || null,
          memory_type: 'episodic',
          primary_abstraction: file.replace(/\.[^.]+$/, '').slice(0, 50),
          cue_anchors: extractAnchors(content),
          memory_value: content.slice(0, 1000),
          energy: 0.5,
          created_at: new Date().toISOString(),
          updated_at: new Date().toISOString()
        };
        // Write to tier2
        const tier2Dir = path.join(memoryDir, 'tier2');
        const goalId = unit.goal_id || '__global__';
        const tierFile = path.join(tier2Dir, `${goalId}.json`);
        const existing = fs.existsSync(tierFile) ? JSON.parse(fs.readFileSync(tierFile, 'utf-8')) : [];
        existing.push(unit);
        fs.writeFileSync(tierFile, JSON.stringify(existing, null, 2), 'utf-8');
        indexManager.addEntry(unit, 'tier2');
        result.migrated++;
      } catch (err: any) {
        result.errors.push(`lessons/${file}: ${err.message}`);
      }
    }
  }

  // 2. Migrate parametric/ → tier3/ (semantic)
  // (Similar pattern)
  // 3. Backup old dirs
  const backupDir = path.join(baseDir, '.backup-v61');
  if (!fs.existsSync(backupDir)) fs.mkdirSync(backupDir, { recursive: true });
  for (const oldDir of ['lessons', 'parametric']) {
    const src = path.join(baseDir, oldDir);
    if (fs.existsSync(src)) {
      fs.renameSync(src, path.join(backupDir, oldDir));
    }
  }

  return result;
}

function extractAnchors(content: string): string[] {
  const anchors = new Set<string>();
  const patterns = [
    /(jwt|oauth|bcrypt|auth)/gi, /(error|fail|exception)/gi,
    /(coverage|test|jest)/gi, /(api|endpoint|route)/gi,
    /(config|setting|env)/gi, /(async|await|promise)/gi
  ];
  for (const pattern of patterns) {
    const matches = content.match(pattern);
    if (matches) matches.forEach(m => anchors.add(m.toLowerCase()));
  }
  return Array.from(anchors).slice(0, 8);
}
```

### Search Implementation (`src/tools/search.ts`)

```typescript
async execute({ goalId, query, maxResults = 10 }: any) {
  // 1. Search harmonic index (tier-agnostic)
  let entries = harmonicIndex.search(query, maxResults * 2);

  // 2. Filter by goalId if specified
  if (goalId) {
    entries = entries.filter(e => e.goal_id === goalId || e.goal_id === null);
  }

  // 3. Load full units from tier files
  const results = [];
  for (const entry of entries.slice(0, maxResults)) {
    const unit = this.loadHarmonicUnit(entry.tier, entry.id);
    if (unit) {
      results.push({
        ...unit,
        score: entry.energy,
        memory_type: entry.memory_type
      });
    }
  }
  return results;
}

private loadHarmonicUnit(tier: string, id: string): HarmonicUnit | null {
  const memoryDir = path.join(this.mafwDir, 'memory', tier);
  if (!fs.existsSync(memoryDir)) return null;

  const files = fs.readdirSync(memoryDir).filter(f => f.endsWith('.json'));
  for (const file of files) {
    try {
      const units: HarmonicUnit[] = JSON.parse(fs.readFileSync(path.join(memoryDir, file), 'utf-8'));
      const found = units.find(u => u.id === id);
      if (found) return found;
    } catch {}
  }
  return null;
}
```

### BM25 Enhancement (`src/memory/harmonic-index.ts`)

Add to `HarmonicIndexManager`:

```typescript
searchBM25(query: string, topK: number = 20): HarmonicIndexEntry[] {
  const tokens = query.toLowerCase().split(/\s+/).filter(t => t.length > 1);
  if (tokens.length === 0) return [];

  const N = this.index.entries.length;
  const df = new Map<string, number>();  // document frequency

  for (const entry of this.index.entries) {
    const text = (entry.primary_abstraction + ' ' + entry.cue_anchors.join(' ')).toLowerCase();
    for (const token of tokens) {
      if (text.includes(token)) df.set(token, (df.get(token) || 0) + 1);
    }
  }

  const scored = this.index.entries.map(entry => {
    const text = (entry.primary_abstraction + ' ' + entry.cue_anchors.join(' ')).toLowerCase();
    let score = 0;
    for (const token of tokens) {
      const tf = (text.match(new RegExp(token.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'g')) || []).length;
      const idf = Math.log((N - (df.get(token) || 0) + 0.5) / ((df.get(token) || 0) + 0.5) + 1);
      score += (tf * (1.2 + 1)) / (tf + 1.2 * (1 - 0.75 + 0.75 * (text.split(/\s+/).length / 10)));
      score += idf;
    }
    return { entry, score: score * entry.energy };
  });

  return scored.sort((a, b) => b.score - a.score).slice(0, topK).map(s => s.entry);
}
```

## Decision Log

| # | Decision | Rationale |
|---|---|---|
| 1 | Cross-tier BM25 only on `primary_abstraction` + `cue_anchors` | Index is compact, search is fast, no need to index full `memory_value` |
| 2 | Old files backed up, never deleted | Safe rollback |
| 3 | No MinHash in M1-M3 | YAGNI — merge comes in M4-M5 |
| 4 | No vector index removal in M1-M3 | Keep `vector-index.ts` in tree but unused; remove in M4 |
| 5 | `@xenova/transformers` retained as optional dep | Breaks nothing, remove when ready |
| 6 | Migration runs once (first load without `.harmonic_index.json`) | Idempotent via existence check |
